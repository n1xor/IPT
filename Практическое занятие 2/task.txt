#include <iostream> // Подключение библиотеки iostream
#include <vector>   // Подключение библиотеки vector

void printMerged(const std::vector<int>& a, const std::vector<int>& b) {
    // Создание процедуры с параметрами, состоящая из указателей на два массива
    size_t idx_a = 0;// Создание счётчика для перемещённых элементов из массива a
    size_t idx_b = 0;// Создание счётчика для перемещённых элементов из массива b

    // 1. Сливаем, пока в обоих массивах есть элементы
    while (idx_a < a.size() && idx_b < b.size()) {
        /*
        Создание цикла while, который будет выполняться до тех пор, пока в обоих массивах
        есть элементы
        */
        
        if (a[idx_a] < b[idx_b]) {
            // Если последний неперемещённый элемент из массива a меньше, чем из массива b,
            // то он выводится на экран и увеличивается значение idx_a на 1 
            std::cout << a[idx_a] << ' ';
            idx_a++;
        }
        else {
            // Если последний неперемещённый элемент из массива a больше, чем из массива b,
            // то он выводится на экран и увеличивается значение idx_и на 1 
            std::cout << b[idx_b] << ' ';
            idx_b++;
        }
    }

    // Если в массиве a остались элементы, выводим их
    while (idx_a < a.size()) {//Цикл длится, пока idx_a меньше размера массива a 
        std::cout << a[idx_a] << ' ';//Вывод на экран
        idx_a++;//Увеличение счётчика на 1
    }

    // 3. Если массиве b остались элементы, выводим их
    while (idx_b < b.size()) {//Цикл длится, пока idx_b меньше размера массива b
        std::cout << b[idx_b] << ' '; //Вывод на экран
        idx_b++;//Увеличение счётчика на 1
    }

    std::cout << '\n';//Вывод для перевода на следующую строку
}

int main() {
    //Точка входа в программу
    std::vector<int> A = { 1, 3, 5, 7 };//Создание массива из целых чисел
    std::vector<int> B = { 2, 4, 6, 8, 10 };//Создание массива из целых чисел

    printMerged(A, B); // Вызов функции

    return 0;// Завершение работы программы
}

#include <iostream>// Подключение библиотеки iostream

using namespace std;// Подключение пространства имён std

int main() {
    //Точка входа в программу
    int n;//Создание числа для объявления края последовательности 
    cin >> n;//Ввод числа n

    // Сумма чисел, которые нам дали
    long long current_sum = 0;//Создание переменной, хранящую сумму

    // В задаче не хватает одного числа, значит, вводить будем n-1 раз
    for (int i = 0; i < n - 1; ++i) {
        int number;//Создаём переменную для нового числа
        cin >> number;//Вводим n
        current_sum += number; // Складываем то, что ввели
    }

    // Сумма всех чисел от 1 до n по формуле Гаусса
    //Приводим переменную n к типу long long для избежания возможного переполнения int
    long long total_sum = (long long)n * (n + 1) / 2;

    // Разница между полной суммой и той, что у нас есть,
    // и будет недостающим числом.
    cout << total_sum - current_sum << endl;//Вывод числа

    return 0;//Завершение работы программы
}

    int a = 4;//Создаём переменную a
    int b = 5;//Создаём переменную b
    a += b;// Увеличиваем значение a на значение переменной b

    a -= b;// Уменьшаем значение a на значение переменной b

    return 0;//Завершение работы программы
}
